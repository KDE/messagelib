macro(add_messagecore_dkim_verify_unittest source)
    get_filename_component(name ${source} NAME_WE)
    ecm_add_test(${source} ${name}.h
        TEST_NAME ${name}
        LINK_LIBRARIES KPim6::MessageCore Qt::Test OpenSSL::SSL
    )
endmacro()
